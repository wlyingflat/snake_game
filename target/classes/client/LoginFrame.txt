package snake.client;

import java.awt.*;
import java.awt.event.*;
import java.io.*;
import java.net.Socket;
import javax.swing.*;
import snake.common.Protocol;

public class LoginFrame extends JFrame {
  private final String serverIp;
  private final int serverPort;
  private JTextField usernameField;
  private JPasswordField passwordField;
  private JButton registerBtn;
  private JButton loginBtn;

  public LoginFrame(String serverIp, int serverPort) {
    this.serverIp = serverIp;
    this.serverPort = serverPort;
    initUI();
  }

  private void initUI() {
    setTitle("Snake Game - Login");
    setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
    setSize(400, 250);
    setLocationRelativeTo(null);
    setResizable(false);

    JPanel panel = new JPanel(new GridBagLayout());
    GridBagConstraints gbc = new GridBagConstraints();
    gbc.insets = new Insets(5, 5, 5, 5);

    // Title
    JLabel titleLabel = new JLabel("=== Snake Game ===");
    titleLabel.setFont(new Font("Arial", Font.BOLD, 20));
    gbc.gridx = 0;
    gbc.gridy = 0;
    gbc.gridwidth = 2;
    panel.add(titleLabel, gbc);

    // Username
    gbc.gridwidth = 1;
    gbc.gridy = 1;
    gbc.gridx = 0;
    panel.add(new JLabel("Username:"), gbc);
    usernameField = new JTextField(15);
    gbc.gridx = 1;
    panel.add(usernameField, gbc);

    // Password
    gbc.gridy = 2;
    gbc.gridx = 0;
    panel.add(new JLabel("Password:"), gbc);
    passwordField = new JPasswordField(15);
    gbc.gridx = 1;
    panel.add(passwordField, gbc);

    // Buttons
    JPanel buttonPanel = new JPanel(new FlowLayout());
    registerBtn = new JButton("Register");
    loginBtn = new JButton("Login");
    buttonPanel.add(registerBtn);
    buttonPanel.add(loginBtn);
    gbc.gridy = 3;
    gbc.gridx = 0;
    gbc.gridwidth = 2;
    panel.add(buttonPanel, gbc);

    add(panel);

    registerBtn.addActionListener(e -> doAuth(true));
    loginBtn.addActionListener(e -> doAuth(false));

    // Enter key triggers login
    getRootPane().setDefaultButton(loginBtn);
  }

  private void doAuth(boolean isRegister) {
    System.out.println("doAuth called, isRegister=" + isRegister);
    String username = usernameField.getText().trim();
    String password = new String(passwordField.getPassword()).trim();
    System.out.println("Username: " + username + ", Password length: " + password.length());
    if (username.isEmpty() || password.isEmpty()) {
      JOptionPane.showMessageDialog(this, "Username and password required.");
      return;
    }

    setEnabled(false);
    new SwingWorker<String, Void>() {
      @Override
      protected String doInBackground() throws Exception {
        return authenticate(username, password, isRegister);
      }

      @Override
      protected void done() {
        setEnabled(true);
        try {
          String result = get();
          System.out.println("Auth result: " + result);
          if (result.startsWith("OK")) {
            String[] lines = result.split("\n");
            String gatewayLine = lines[1];
            String[] parts = gatewayLine.split(" ");
            String gatewayIp = parts[1];
            int gatewayPort = Integer.parseInt(parts[2]);
            System.out.println("Gateway: " + gatewayIp + ":" + gatewayPort);
            // 尝试创建房间列表窗口，捕获异常
            try {
              RoomListFrame frame =
                  new RoomListFrame(username, gatewayIp, gatewayPort, serverIp, serverPort);
              frame.setVisible(true);
              dispose();
            } catch (Exception e) {
              e.printStackTrace();
              JOptionPane.showMessageDialog(
                  LoginFrame.this, "Failed to open room list: " + e.getMessage());
            }
          } else {
            JOptionPane.showMessageDialog(LoginFrame.this, result);
          }
        } catch (Exception ex) {
          ex.printStackTrace();
          JOptionPane.showMessageDialog(LoginFrame.this, "Error: " + ex.getMessage());
        }
      }
    }.execute();
  }

  private String authenticate(String username, String password, boolean isRegister)
      throws IOException {
    String cmd = isRegister ? Protocol.CMD_REGISTER : Protocol.CMD_LOGIN;
    System.out.println("Sending: " + cmd + " " + username + " " + password);
    try (Socket socket = new Socket(serverIp, serverPort);
        PrintWriter out = new PrintWriter(socket.getOutputStream(), true);
        BufferedReader in = new BufferedReader(new InputStreamReader(socket.getInputStream()))) {
      out.println(cmd + " " + username + " " + password);
      System.out.println("Sent, waiting for response...");
      StringBuilder sb = new StringBuilder();
      String line;
      while ((line = in.readLine()) != null) {
        sb.append(line).append("\n");
        System.out.println("Received line: " + line);
      }
      String result = sb.toString().trim();
      System.out.println("Full response: " + result);
      return result;
    }
  }
}
