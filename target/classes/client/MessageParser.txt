package snake.client;

import com.fasterxml.jackson.databind.ObjectMapper;
import java.io.IOException;
import java.time.LocalTime;
import java.time.format.DateTimeFormatter;
import snake.common.*;

public class MessageParser {
  private static final ObjectMapper mapper = new ObjectMapper();
  private static final DateTimeFormatter TIME_FORMAT = DateTimeFormatter.ofPattern("HH:mm:ss.SSS");

  private static void log(String msg) {
    System.out.println("[" + LocalTime.now().format(TIME_FORMAT) + "] [Parser] " + msg);
  }

  public static GameStateData deserializeGameState(String json) {
    log(
        "Parsing JSON: "
            + (json != null ? json.substring(0, Math.min(200, json.length())) : "null"));
    if (json == null || json.trim().isEmpty()) {
      log("Empty or null input");
      return null;
    }
    try {
      var root = mapper.readTree(json);
      if (!root.has("type") || !"STATE".equals(root.get("type").asText())) {
        log(
            "Not a STATE message, type="
                + (root.has("type") ? root.get("type").asText() : "missing"));
        return null;
      }
      var dataNode = root.get("data");
      if (dataNode == null) {
        log("Missing 'data' field");
        return null;
      }
      GameStateData state = mapper.treeToValue(dataNode, GameStateData.class);
      log("Parsed successfully: roomId=" + state.roomId + ", players=" + state.playerCount);
      return state;
    } catch (IOException e) {
      log("Parse error: " + e.getMessage());
      e.printStackTrace();
      return null;
    }
  }
}
